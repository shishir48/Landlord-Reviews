# Landlord Reviews — Backend Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Node.js/Express/MongoDB REST API that powers the Landlord Reviews mobile app, including Firebase auth middleware, property/review/landlord CRUD, and an admin landlord-merge dashboard.

**Architecture:** Express app with Firebase Admin SDK token verification on every route. MongoDB via Mongoose. `avg_rating` and `review_count` denormalized on Property and updated atomically on each review write. Landlord dedup via Levenshtein fuzzy search (tenant side) and admin merge endpoint (sets `merged_into` + re-points property refs).

**Tech Stack:** Node.js 20, Express 4, Mongoose 8, Firebase Admin SDK 12, Jest 29, Supertest, mongodb-memory-server 9

```js
const admin = require('../config/firebase');

async function verifyToken(req, res, next) {
  const header = req.headers.authorization;
  if (!header?.startsWith('Bearer ')) {
    return res.status(401).json({ error: 'Missing token' });
  }
  const token = header.split(' ')[1];
  try {
    const decoded = await admin.auth().verifyIdToken(token);
    req.user = decoded;
    next();
  } catch {
    res.status(401).json({ error: 'Invalid token' });
  }
}

module.exports = { verifyToken };
```

```js
function levenshtein(a, b) {
  a = a.toLowerCase();
  b = b.toLowerCase();
  const dp = Array.from({ length: a.length + 1 }, (_, i) =>
    Array.from({ length: b.length + 1 }, (_, j) => (i === 0 ? j : j === 0 ? i : 0))
  );
  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= b.length; j++) {
      dp[i][j] = a[i - 1] === b[j - 1]
        ? dp[i - 1][j - 1]
        : 1 + Math.min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1]);
    }
  }
  return dp[a.length][b.length];
}

module.exports = { levenshtein };
```

```js
const { Router } = require('express');
const { verifyToken } = require('../middleware/auth');
const Property = require('../models/Property');
const { normalizeAddress } = require('../utils/normalizeAddress');

const router = Router();
router.use(verifyToken);

router.post('/', async (req, res) => {
  const { place_id, formatted_address, lat, lng } = req.body;
  const normalized_address = normalizeAddress(formatted_address);

  const filter = place_id ? { place_id } : { normalized_address };
  const existing = await Property.findOne(filter);
  if (existing) return res.status(200).json({ property: existing });

  const property = await Property.create({
    place_id: place_id || null,
    formatted_address,
    normalized_address,
    lat,
    lng,
    createdBy: req.user.uid,
  });
  res.status(201).json({ property });
});

router.get('/', async (req, res) => {
  const { place_id } = req.query;
  if (!place_id) return res.status(400).json({ error: 'place_id required' });

  const property = await Property.findOne({ place_id }).populate('landlord_id');
  if (!property) return res.status(404).json({ error: 'Not found' });
  res.json({ property });
});

module.exports = router;
```

```js
const { Router } = require('express');
const mongoose = require('mongoose');
const { verifyToken } = require('../middleware/auth');
const Review = require('../models/Review');
const Property = require('../models/Property');

const router = Router();
router.use(verifyToken);

router.post('/', async (req, res) => {
  const { property_id, rating, text, tenancy_period } = req.body;
  const user_id = req.user.uid;

  try {
    const review = await Review.create({ property_id, user_id, rating, text, tenancy_period });

    // Recalculate avg_rating atomically using aggregation
    const [agg] = await Review.aggregate([
      { $match: { property_id: new mongoose.Types.ObjectId(property_id) } },
      { $group: { _id: null, avg: { $avg: '$rating' }, count: { $sum: 1 } } },
    ]);
    await Property.findByIdAndUpdate(property_id, {
      avg_rating: Math.round(agg.avg * 10) / 10,
      review_count: agg.count,
    });

    res.status(201).json({ review });
  } catch (err) {
    if (err.code === 11000) return res.status(409).json({ error: 'Already reviewed' });
    throw err;
  }
});

router.get('/', async (req, res) => {
  const { property_id } = req.query;
  if (!property_id) return res.status(400).json({ error: 'property_id required' });
  const reviews = await Review.find({ property_id }).sort({ createdAt: -1 });
  res.json({ reviews });
});

router.put('/:id', async (req, res) => {
  const review = await Review.findOne({ _id: req.params.id, user_id: req.user.uid });
  if (!review) return res.status(404).json({ error: 'Not found' });

  review.rating = req.body.rating ?? review.rating;
  review.text = req.body.text ?? review.text;
  review.tenancy_period = req.body.tenancy_period ?? review.tenancy_period;
  await review.save();

  const [agg] = await Review.aggregate([
    { $match: { property_id: review.property_id } },
    { $group: { _id: null, avg: { $avg: '$rating' }, count: { $sum: 1 } } },
  ]);
  await Property.findByIdAndUpdate(review.property_id, {
    avg_rating: Math.round(agg.avg * 10) / 10,
    review_count: agg.count,
  });

  res.json({ review });
});
```

```js
const { Router } = require('express');
const mongoose = require('mongoose');
const { verifyToken } = require('../middleware/auth');
const { adminOnly } = require('../middleware/adminOnly');
const Landlord = require('../models/Landlord');
const Property = require('../models/Property');
const Review = require('../models/Review');
const { levenshtein } = require('../utils/levenshtein');

const router = Router();
router.use(verifyToken, adminOnly);

const DUPLICATE_THRESHOLD = 4;

router.get('/landlords/duplicates', async (_req, res) => {
  const landlords = await Landlord.find({ merged_into: null }).select('name');
  const pairs = [];
  for (let i = 0; i < landlords.length; i++) {
    for (let j = i + 1; j < landlords.length; j++) {
      const dist = levenshtein(landlords[i].name, landlords[j].name);
      if (dist <= DUPLICATE_THRESHOLD) {
        pairs.push({ a: landlords[i], b: landlords[j], distance: dist });
      }
    }
  }
  pairs.sort((x, y) => x.distance - y.distance);
  res.json({ pairs });
});

router.post('/landlords/merge', async (req, res) => {
  const { canonicalId, duplicateIds } = req.body;
  const dupObjectIds = duplicateIds.map(id => new mongoose.Types.ObjectId(id));

  await Landlord.updateMany(
    { _id: { $in: dupObjectIds } },
    { $set: { merged_into: new mongoose.Types.ObjectId(canonicalId) } }
  );
  await Property.updateMany(
    { landlord_id: { $in: dupObjectIds } },
    { $set: { landlord_id: new mongoose.Types.ObjectId(canonicalId) } }
  );

  res.json({ merged: duplicateIds.length });
});

router.get('/properties', async (_req, res) => {
  const properties = await Property.find().populate('landlord_id').sort({ createdAt: -1 });
  res.json({ properties });
});

router.delete('/reviews/:id', async (req, res) => {
  const review = await Review.findByIdAndDelete(req.params.id);
  if (!review) return res.status(404).json({ error: 'Not found' });

  const agg = await Review.aggregate([
    { $match: { property_id: review.property_id } },
    { $group: { _id: null, avg: { $avg: '$rating' }, count: { $sum: 1 } } },
  ]);
  await Property.findByIdAndUpdate(review.property_id, {
    avg_rating: agg[0] ? Math.round(agg[0].avg * 10) / 10 : 0,
    review_count: agg[0]?.count ?? 0,
  });

  res.status(204).send();
});

module.exports = router;
```
